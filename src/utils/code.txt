'use client';

import { useState, useEffect, useRef, ComponentType } from 'react';
import dynamic from 'next/dynamic';

type LazyLoaderProps = {
  loader: () => Promise<{ default: ComponentType<any> }>;
  height?: number; // reserve space to prevent CLS
  fallback?: React.ReactNode;
};

/**
 * Generic intersection observer loader
 * Dynamically imports a client component when it becomes visible.
 */
export default function LazyLoader({
  loader,
  height = 300,
  fallback = <div style={{ height }}>Loading…</div>,
}: LazyLoaderProps) {
  const [show, setShow] = useState(false);
  const ref = useRef<HTMLDivElement>(null);

  // Only create dynamic component after visibility
  const [DynamicComp, setDynamicComp] = useState<ComponentType<any> | null>(null);

  useEffect(() => {
    const observer = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) {
        setShow(true);
        observer.disconnect();
      }
    });
    if (ref.current) observer.observe(ref.current);
    return () => observer.disconnect();
  }, []);

  // Dynamically import once visible
  useEffect(() => {
    if (show) {
      const Dyn = dynamic(loader, { ssr: false });
      setDynamicComp(() => Dyn);
    }
  }, [show, loader]);

  return <div ref={ref}>{DynamicComp ? <DynamicComp /> : fallback}</div>;
}

export function useHighlightUnderline(selector = '.highlighted-text') {
  useEffect(() => {
    // run in idle time, post-paint
    const run = () => {
      const els = document.querySelectorAll<HTMLElement>(selector)
      els.forEach((el) => {
        // Skip if already processed
        if (el.dataset.svgApplied) return
        el.dataset.svgApplied = 'true'

        // apply background as data URI (async to avoid main-thread block)
        requestIdleCallback?.(() => {
          el.style.setProperty(
            '--underline-svg',
            `url("data:image/svg+xml;utf8,<svg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 400 20' preserveAspectRatio='none'><path d='M0 10 Q200 25 400 10' stroke='%23e60028' stroke-width='4' fill='none'/></svg>")`
          )
          el.style.backgroundImage = 'var(--underline-svg)'
          const pseudo = window.getComputedStyle(el, '::after')
          // Trigger fade-in
          el.style.setProperty('--underline-opacity', '1')
          el.classList.add('underline-ready')
        })
      })
    }

    // Use requestIdleCallback or fallback to timeout
    if ('requestIdleCallback' in window) {
      requestIdleCallback(run, { timeout: 2000 })
    } else {
      setTimeout(run, 500)
    }
  }, [selector])
}

'use client'

import { useEffect } from 'react'

export function useHighlightUnderline(
  selector = '.highlighted-text',
  svgPath = '/assets/underline.svg'
) {
  useEffect(() => {
    const applyUnderline = () => {
      const elements = document.querySelectorAll<HTMLElement>(selector)
      elements.forEach((el) => {
        if (el.dataset.underlineApplied) return // avoid duplicate
        el.dataset.underlineApplied = 'true'

        // Apply underline lazily in an idle frame (non-blocking)
        requestIdleCallback?.(() => {
          // Use inline style so it doesn’t trigger layout reflow globally
          el.style.setProperty('--underline-image', `url(${svgPath})`)
          el.style.setProperty('--underline-opacity', '1')

          // Apply background image via pseudo-element
          el.style.setProperty('position', 'relative')
          el.style.setProperty('--underline-opacity', '1')

          // Directly set ::after background
          const sheet = document.createElement('style')
          sheet.textContent = `
            [data-underline-applied="true"]::after {
              background-image: url(${svgPath});
              opacity: 1;
            }
          `
          document.head.appendChild(sheet)
        })
      })
    }

    // Run after LCP to avoid blocking paint
    if ('requestIdleCallback' in window) {
      requestIdleCallback(applyUnderline, { timeout: 2000 })
    } else {
      setTimeout(applyUnderline, 600)
    }
  }, [selector, svgPath])
}

// app/api/pdf/route.ts
import { NextResponse } from "next/server";
import fs from "fs";
import path from "path";

export async function GET() {
  const filePath = path.join(process.cwd(), "protected", "my.pdf");
  const fileBuffer = fs.readFileSync(filePath);

  // Set headers to prevent download
  const response = new NextResponse(fileBuffer);
  response.headers.set("Content-Type", "application/pdf");
  response.headers.set("Content-Disposition", "inline"); // ensures it opens inline, not as a download
  response.headers.set("Cache-Control", "no-store");

  // Optional: block save-as in some browsers (not foolproof)
  response.headers.set(
    "Content-Security-Policy",
    "default-src 'self'; frame-ancestors 'none';"
  );

  return response;
}


// app/api/pdf/route.ts
export const runtime = "edge";

export async function GET(req: Request) {
  const referer = req.headers.get("referer") || "";

  if (!referer.startsWith("https://yourdomain.com/docs")) {
    return new Response("Forbidden", { status: 403 });
  }

  const pdfResponse = await fetch(`${req.url.origin}/_next/static/my.pdf`);
  if (!pdfResponse.ok) return new Response("Not found", { status: 404 });

  const arrayBuffer = await pdfResponse.arrayBuffer();

  return new Response(arrayBuffer, {
    headers: {
      "Content-Type": "application/pdf",
      "Content-Disposition": "inline",
      "Cache-Control": "no-store",
    },
  });
}
