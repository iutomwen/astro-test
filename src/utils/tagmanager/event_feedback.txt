import { sqliteTable, text, integer } from "drizzle-orm/sqlite-core";

/**
 * EVENT
 */
export const event = sqliteTable("event", {
  id: text("id").primaryKey(), // uuid
  name: text("name").notNull(),
  date: integer("date", { mode: "timestamp" }).notNull(),
  location: text("location"),
});

/**
 * EVENT_USER (one per user per event)
 */
export const eventUser = sqliteTable("event_user", {
  id: text("id").primaryKey(), // uuid
  eventId: text("event_id")
    .notNull()
    .references(() => event.id),
  userId: text("user_id").notNull(),
  createdAt: integer("created_at", { mode: "timestamp" }).notNull(),
});

/**
 * APPOINTMENT
 */
export const appointment = sqliteTable("appointment", {
  id: text("id").primaryKey(),
  eventUserId: text("event_user_id")
    .notNull()
    .references(() => eventUser.id),
  bookedAt: integer("booked_at", { mode: "timestamp" }).notNull(),
});

/**
 * EVENT FEEDBACK (NEW)
 */

import { NextRequest, NextResponse } from "next/server";
import { drizzle } from "drizzle-orm/d1";
import { eventFeedback } from "@/db/schema";
import { eq } from "drizzle-orm";
import { v4 as uuid } from "uuid";

export async function POST(req: NextRequest) {
  const body = await req.json();

  const {
    eventUserId,
    overallRating,
    speakerRating,
    venueRating,
    organisationRating,
    comments,
  } = body;

  if (!eventUserId) {
    return NextResponse.json({ error: "Missing eventUserId" }, { status: 400 });
  }

  const db = drizzle(process.env.DB as any);

  // Prevent duplicate feedback
  const existing = await db
    .select()
    .from(eventFeedback)
    .where(eq(eventFeedback.eventUserId, eventUserId))
    .get();

  if (existing) {
    return NextResponse.json(
      { error: "Feedback already submitted" },
      { status: 400 }
    );
  }

  await db.insert(eventFeedback).values({
    id: uuid(),
    eventUserId,
    overallRating,
    speakerRating,
    venueRating,
    organisationRating,
    comments,
    createdAt: new Date(),
  });

  return NextResponse.json({ success: true });
}

"use client";

import { useState } from "react";

export default function FeedbackPage({
  params,
}: {
  params: { eventUserId: string };
}) {
  const [loading, setLoading] = useState(false);

  const [form, setForm] = useState({
    overallRating: 5,
    speakerRating: 5,
    venueRating: 5,
    organisationRating: 5,
    comments: "",
  });

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);

    await fetch("/api/feedback", {
      method: "POST",
      body: JSON.stringify({
        ...form,
        eventUserId: params.eventUserId,
      }),
    });

    setLoading(false);
    alert("Thank you for your feedback!");
  };

  const RatingSelect = ({
    label,
    value,
    onChange,
  }: {
    label: string;
    value: number;
    onChange: (v: number) => void;
  }) => (
    <div className="space-y-2">
      <label className="font-medium">{label}</label>
      <select
        className="w-full border rounded p-2"
        value={value}
        onChange={(e) => onChange(Number(e.target.value))}
      >
        {[1, 2, 3, 4, 5].map((n) => (
          <option key={n} value={n}>
            {n} ⭐
          </option>
        ))}
      </select>
    </div>
  );

  return (
    <div className="max-w-xl mx-auto p-6 space-y-6">
      <h1 className="text-2xl font-bold">Event Feedback</h1>

      <form onSubmit={handleSubmit} className="space-y-4">
        <RatingSelect
          label="Overall Experience"
          value={form.overallRating}
          onChange={(v) => setForm({ ...form, overallRating: v })}
        />

        <RatingSelect
          label="Speaker"
          value={form.speakerRating}
          onChange={(v) => setForm({ ...form, speakerRating: v })}
        />

        <RatingSelect
          label="Venue"
          value={form.venueRating}
          onChange={(v) => setForm({ ...form, venueRating: v })}
        />

        <RatingSelect
          label="Organisation"
          value={form.organisationRating}
          onChange={(v) => setForm({ ...form, organisationRating: v })}
        />

        <div>
          <label className="font-medium">Additional Comments</label>
          <textarea
            className="w-full border rounded p-2 mt-2"
            rows={4}
            value={form.comments}
            onChange={(e) =>
              setForm({ ...form, comments: e.target.value })
            }
          />
        </div>

        <button
          disabled={loading}
          className="w-full bg-black text-white py-2 rounded"
        >
          {loading ? "Submitting..." : "Submit Feedback"}
        </button>
      </form>
    </div>
  );
}

function classifyNPS(score: number) {
  if (score >= 9) return "promoter";
  if (score >= 7) return "passive";
  return "detractor";
}

nps calculator
import { sql } from "drizzle-orm";

const results = await db
  .select({
    promoters: sql<number>`sum(case when ${eventFeedback.npsScore} >= 9 then 1 else 0 end)`,
    passives: sql<number>`sum(case when ${eventFeedback.npsScore} between 7 and 8 then 1 else 0 end)`,
    detractors: sql<number>`sum(case when ${eventFeedback.npsScore} <= 6 then 1 else 0 end)`,
    total: sql<number>`count(*)`,
  })
  .from(eventFeedback)
  .innerJoin(eventUser, eq(eventUser.id, eventFeedback.eventUserId))
  .where(eq(eventUser.eventId, eventId))
  .get();

const nps =
  ((results.promoters - results.detractors) / results.total) * 100;
